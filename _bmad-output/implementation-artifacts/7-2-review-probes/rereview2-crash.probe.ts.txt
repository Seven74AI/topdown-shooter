import { randomUUID } from "node:crypto";
import pg from "pg";
import { accountsRepo } from "../accountsRepo";
import { openPostgres } from "../client";
import { migrateDb } from "../migrate";
import { progressRepo } from "../progressRepo";
import { log } from "../../ops/logger";
log.level = "silent";
const URL = process.env.TEST_DATABASE_URL!;
const database = openPostgres(URL);
await migrateDb({ kind: "postgres", db: await database.db() });
const a = await accountsRepo(database).createGuest(randomUUID().replace(/-/g, "").padEnd(64, "a"), "Ana");
const repo = progressRepo(database);
await repo.addSupplies(randomUUID(), a.id, () => ({ supplies: 1000, breakdown: {} }));
const lock = new pg.Client({ connectionString: URL }); await lock.connect();
await lock.query("begin"); await lock.query("select * from progress where account_id=$1 for update", [a.id]);
const p = repo.addSupplies(randomUUID(), a.id, () => ({ supplies: 5, breakdown: {} })).then(() => "resolved", (e) => "rejected " + e.message);
await new Promise((r) => setTimeout(r, 400));
const admin = new pg.Client({ connectionString: URL }); await admin.connect();
const k = await admin.query(`select pg_terminate_backend(pid) from pg_stat_activity where wait_event_type='Lock' and datname=current_database()`);
console.log("killed", k.rowCount, "→", await p);
await new Promise((r) => setTimeout(r, 500));
console.log("STILL ALIVE");
await lock.query("rollback"); await lock.end(); await admin.end(); await database.close();
